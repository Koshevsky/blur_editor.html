""" bot_blur_patch.py URL Telegram WebApp button limit: ~4096 chars total. Strategy: - Base URL + ?dur=X ≈ 80 chars - Hash #img=B64 ≈ remaining budget - We target ≤3900 chars for the base64 payload to stay safe. - Extract a tiny frame (120px wide), compress aggressively until it fits. - Editor has scrubber but shows only one reference frame. """ import os import json import uuid import base64 import shutil import asyncio import logging import subprocess import urllib.parse from aiogram import Router, types, F from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton, WebAppInfo WEBAPP_URL = "https://koshevsky.github.io/blur_editor.html/blur_editor.html" FFMPEG_PATH = r"E:\ffmpeg\ffmpeg.exe" TEMP_DIR = "temp" BLUR_QUEUE_DIR = os.path.join(TEMP_DIR, "blur_queue") os.makedirs(BLUR_QUEUE_DIR, exist_ok=True) router = Router() _awaiting_blur: dict[int, dict] = {} # Telegram inline button URL hard limit (with safe margin) TG_URL_LIMIT = 4000 def _ffprobe_duration(path: str) -> float: ffprobe = FFMPEG_PATH.replace("ffmpeg.exe", "ffprobe.exe").replace("ffmpeg", "ffprobe") if not os.path.exists(ffprobe): ffprobe = "ffprobe" r = subprocess.run( [ffprobe, "-v", "error", "-show_entries", "format=duration", "-of", "default=noprint_wrappers=1:nokey=1", path], capture_output=True, text=True, ) try: return max(float(r.stdout.strip()), 0.1) except Exception: return 30.0 def _extract_frame_b64_sized(video_path: str, t: float, max_encoded_chars: int) -> str: """ Извлекает кадр, подбирает ширину и качество пока URL-encoded base64 не войдёт в max_encoded_chars символов. """ tmp = os.path.join(TEMP_DIR, f"frame_{uuid.uuid4().hex[:8]}.jpg") # От менее сжатого к более сжатому attempts = [ (160, 15), (130, 18), (110, 22), (90, 25), (75, 28), (60, 30), ] try: for width, quality in attempts: r = subprocess.run( [FFMPEG_PATH, "-y", "-ss", f"{t:.3f}", "-i", video_path, "-vframes", "1", "-vf", f"scale={width}:-2", "-q:v", str(quality), tmp], capture_output=True, ) if r.returncode != 0 or not os.path.exists(tmp): continue with open(tmp, "rb") as f: raw = f.read() b64 = base64.b64encode(raw).decode() encoded = urllib.parse.quote(b64, safe="") if len(encoded) <= max_encoded_chars: logging.info( "Blur preview: %dpx q=%d raw=%dB b64=%d encoded=%d (budget=%d)", width, quality, len(raw), len(b64), len(encoded), max_encoded_chars ) return b64 logging.warning("Blur preview: no frame fit in budget=%d", max_encoded_chars) return "" except Exception as e: logging.warning("frame extract failed: %s", e) return "" finally: if os.path.exists(tmp): try: os.remove(tmp) except OSError: pass def build_webapp_url(video_path: str) -> tuple[str, float]: """ Строит URL для WebApp. Гарантирует len(url) <= TG_URL_LIMIT. Кадр превью пакуется в URL hash (#img=...) — hash не логируется Telegram. """ dur = _ffprobe_duration(video_path) base_url = f"{WEBAPP_URL}?dur={dur:.1f}" # "#img=" = 5 chars budget = TG_URL_LIMIT - len(base_url) - 5 if budget < 100: logging.warning("Base URL already too long (%d chars), no preview", len(base_url)) return base_url, dur t = max(0.05, dur * 0.25) b64 = _extract_frame_b64_sized(video_path, t, budget) if b64: encoded = urllib.parse.quote(b64, safe="") full_url = base_url + "#img=" + encoded logging.info("Blur editor URL: %d chars (limit %d)", len(full_url), TG_URL_LIMIT) return full_url, dur logging.info("Blur editor URL (no preview): %d chars", len(base_url)) return base_url, dur def apply_manual_blur(src: str, dst: str, regions: list, blur_radius: int = 18, settings: dict = None): if not regions: shutil.copy2(src, dst) return ffprobe = FFMPEG_PATH.replace("ffmpeg.exe", "ffprobe.exe").replace("ffmpeg", "ffprobe") if not os.path.exists(ffprobe): ffprobe = "ffprobe" r = subprocess.run( [ffprobe, "-v", "error", "-select_streams", "v:0", "-show_entries", "stream=width,height", "-of", "csv=p=0", src], capture_output=True, text=True, ) try: vid_w, vid_h = map(int, r.stdout.strip().split(",")) except Exception: vid_w, vid_h = 1080, 1920 dur = _ffprobe_duration(src) parts = [] current = "[0:v]" for i, reg in enumerate(regions): x1 = max(0, int(reg["x1"] * vid_w)) y1 = max(0, int(reg["y1"] * vid_h)) x2 = min(vid_w, int(reg["x2"] * vid_w)) y2 = min(vid_h, int(reg["y2"] * vid_h)) w = max(2, ((x2 - x1) // 2) * 2) h = max(2, ((y2 - y1) // 2) * 2) t0 = float(reg.get("t0", 0.0)) t1 = float(reg.get("t1", dur)) enable = f"between(t,{t0:.3f},{t1:.3f})" out_lbl = "[outv]" if i == len(regions) - 1 else f"[out{i}]" parts.append( f"{current}split=2[main{i}][copy{i}];" f"[copy{i}]crop={w}:{h}:{x1}:{y1}," f"boxblur={blur_radius}:{blur_radius}[blurred{i}];" f"[main{i}][blurred{i}]overlay={x1}:{y1}:enable='{enable}'{out_lbl}" ) current = f"[out{i}]" vf = ";".join(parts) codec, enc_params = "libx264", ["-preset", "fast"] if settings: try: from bot import resolve_video_encoder codec, enc_params, _ = resolve_video_encoder(settings) except Exception: pass def run_cmd(c, ep): return subprocess.run( [FFMPEG_PATH, "-y", "-i", src, "-filter_complex", vf, "-map", "[outv]", "-map", "0:a?", "-c:v", c, *ep, "-pix_fmt", "yuv420p", "-c:a", "copy", dst], capture_output=True, text=True, ) res = run_cmd(codec, enc_params) if res.returncode != 0 and codec != "libx264": logging.warning("GPU blur failed, CPU fallback") res = run_cmd("libx264", ["-preset", "fast"]) if res.returncode != 0: raise RuntimeError(f"FFmpeg error:\n{res.stderr[-400:]}") async def offer_manual_blur(message: types.Message, video_path: str, settings: dict): uid = message.from_user.id queued = os.path.join(BLUR_QUEUE_DIR, f"{uid}_{uuid.uuid4().hex[:8]}.mp4") try: shutil.copy2(video_path, queued) except Exception as e: logging.warning("blur queue copy failed: %s", e) return status = await message.answer("⏳ Готовлю редактор блюра…") try: loop = asyncio.get_event_loop() full_url, dur = await loop.run_in_executor(None, build_webapp_url, queued) except Exception as e: logging.warning("build_webapp_url failed: %s", e) full_url = f"{WEBAPP_URL}?dur=30" finally: try: await status.delete() except Exception: pass _awaiting_blur[uid] = {"video_path": queued, "settings": settings} kb = InlineKeyboardMarkup(inline_keyboard=[[ InlineKeyboardButton( text="✏️ Указать область блюра", web_app=WebAppInfo(url=full_url), ) ]]) await message.answer( "🎯 Хочешь вручную заблюрить область?\n" "Открой редактор → нарисуй прямоугольники → нажми «Применить».", reply_markup=kb, ) @router.message(F.web_app_data) async def handle_blur_webapp_data(message: types.Message): uid = message.from_user.id raw = message.web_app_data.data if message.web_app_data else None if not raw: return try: payload = json.loads(raw) except json.JSONDecodeError: await message.answer("⚠️ Некорректные данные от редактора.") return regions = payload.get("regions", []) if not regions: await message.answer("Областей не выделено — ничего не делаем.") return context = _awaiting_blur.pop(uid, None) if not context: await message.answer("⚠️ Видео не найдено. Обработай ролик заново.") return src = context["video_path"] if not os.path.exists(src): await message.answer("⚠️ Файл видео исчез. Обработай ролик заново.") return dst = src.replace(".mp4", "_blurred.mp4") n = len(regions) status = await message.answer( f"⏳ Применяю блюр к {n} {'области' if n == 1 else 'областям'}…" ) try: loop = asyncio.get_event_loop() await loop.run_in_executor( None, apply_manual_blur, src, dst, regions, 18, context.get("settings") ) await status.delete() await message.answer_video( types.FSInputFile(dst), caption=f"✅ Блюр применён к {n} {'области' if n == 1 else 'областям'}!", ) except Exception as e: logging.error("Manual blur failed: %s", e) await status.delete() await message.answer(f"⚠️ Ошибка блюра:\n{str(e)[:300]}") finally: for p in [src, dst]: try: if p and os.path.exists(p): os.remove(p) except OSError: pass
